# PyJWT, not python-jose. python-jose pulled in ecdsa, which carries
# CVE-2024-23342 (Minerva timing attack on P-256). It has no fix and the ecdsa
# maintainers consider side-channel attacks out of scope, so pip-audit failed
# CI on every push with nothing to upgrade to. This service signs with HS256,
# which is HMAC and never touches ECDSA, so the advisory was unreachable, but
# an unfixable permanent CI failure is not worth carrying. PyJWT has zero
# dependencies and removes python-jose, ecdsa, rsa, pyasn1, six, cryptography,
# cffi and pycparser from the tree.
PyJWT==2.14.0

# bcrypt is held at 4.0.1 on purpose. passlib 1.7.4 runs an internal self-test
# on first use that hashes a password longer than 72 bytes. bcrypt 4.1 and
# later refuse that instead of silently truncating, so passlib raises
# "ValueError: password cannot be longer than 72 bytes" and every hash call
# fails. Verified against bcrypt 5.0.0. The fix is to drop passlib and call
# the bcrypt library directly, not to bump this number.
bcrypt==4.0.1
